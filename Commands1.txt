app.py
from flask import Flask
app = Flask(__name__)
@app.route("/")
def home():
    return "AI App Running"
if __name__ == "__main__":
    app.run(host="0.0.0.0", port=5000)

deployment.yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: ai-app
spec:
  replicas: 2
  selector:
    matchLabels:
      app: ai-app
  template:
    metadata:
      labels:
        app: ai-app
    spec:
      containers:
      - name: ai-container
        image: ai-model
        imagePullPolicy: Never
        ports:
        - containerPort: 5000

Dockerfile
FROM python:3.9
WORKDIR /app
COPY . .
RUN pip install -r requirements.txt
CMD ["python", "app.py"]

service.yaml
apiVersion: v1
kind: Service
metadata:
  name: ai-service
spec:
  type: NodePort
  selector:
    app: ai-app
  ports:
  - port: 5000
    targetPort: 5000
    nodePort: 30007
